# ─── TOOL IMPLEMENTATIONS ──────────────────────────

@server.call_tool()
async def call_tool(
    name: str,
    arguments: dict
) -> list[types.TextContent]:
    """Handle tool calls from the AI"""

    # ── Weather ────────────────────────────────────
    if name == "get_weather":
        city = arguments["city"]
        try:
            url = f"https://wttr.in/{city}?format=j1"
            r = requests.get(url, timeout=10)
            d = r.json()["current_condition"][0]

            result = f"""Weather in {city}:
🌡️ Temperature: {d['temp_C']}°C
🤔 Feels like: {d['FeelsLikeC']}°C
☁️  Condition: {d['weatherDesc'][0]['value']}
💧 Humidity: {d['humidity']}%
💨 Wind: {d['windspeedKmph']} km/h
📅 Updated: {datetime.datetime.now().strftime('%H:%M')}"""

        except Exception as e:
            result = f"Could not get weather for {city}: {e}"

        return [types.TextContent(type="text", text=result)]

    # ── Tech News ──────────────────────────────────
    elif name == "get_tech_news":
        topic = arguments["topic"]
        count = arguments.get("count", 5)
        try:
            url = f"https://hn.algolia.com/api/v1/search?query={topic}&hitsPerPage={count}&tags=story"
            r = requests.get(url, timeout=10)
            hits = r.json().get("hits", [])

            news_lines = [f"Latest news about '{topic}':"]
            for i, h in enumerate(hits, 1):
                title = h.get("title", "")
                points = h.get("points", 0)
                url_link = h.get("url", "")
                news_lines.append(f"{i}. {title} ({points} pts)")
                if url_link:
                    news_lines.append(f"   Link: {url_link}")

            result = "\\n".join(news_lines)

        except Exception as e:
            result = f"Could not get news: {e}"

        return [types.TextContent(type="text", text=result)]

    # ── Calculator ─────────────────────────────────
    elif name == "calculate":
        expression = arguments["expression"]
        try:
            import math
            safe_dict = {
                "math": math,
                "sqrt": math.sqrt,
                "pi": math.pi,
                "abs": abs,
                "round": round
            }
            result_val = eval(expression, {"__builtins__": {}}, safe_dict)
            result = f"Calculation: {expression} = {result_val}"
        except Exception as e:
            result = f"Calculation error: {e}"

        return [types.TextContent(type="text", text=result)]

    # ── Lead Qualifier ─────────────────────────────
    elif name == "qualify_lead":
        name = arguments["name"]
        company = arguments.get("company", "Unknown")
        message = arguments["message"]
        has_budget = arguments.get("budget", False)

        # Simple scoring without AI (MCP server should be fast)
        score = 50
        reasons = []

        # Score based on message keywords
        high_intent = ["urgent", "asap", "immediately", "budget approved",
                      "ready to start", "need help now", "decision made"]
        medium_intent = ["interested", "considering", "exploring",
                        "want to learn", "looking into"]
        low_intent = ["just browsing", "curious", "maybe", "not sure",
                     "student", "learning"]

        message_lower = message.lower()

        for keyword in high_intent:
            if keyword in message_lower:
                score += 15
                reasons.append(f"High intent: '{keyword}'")
                break

        for keyword in medium_intent:
            if keyword in message_lower:
                score += 5
                reasons.append(f"Medium intent: '{keyword}'")
                break

        for keyword in low_intent:
            if keyword in message_lower:
                score -= 20
                reasons.append(f"Low intent: '{keyword}'")
                break

        if has_budget:
            score += 20
            reasons.append("Has confirmed budget")

        if company and company != "Unknown":
            score += 10
            reasons.append("Company information provided")

        score = max(0, min(100, score))

        if score >= 70:
            status = "QUALIFIED"
            action = "Schedule demo call within 24 hours"
        elif score >= 50:
            status = "POTENTIAL"
            action = "Add to nurture email sequence"
        else:
            status = "UNQUALIFIED"
            action = "Archive for 3 months follow-up"

        result = f"""Lead Qualification Result:
Name: {name}
Company: {company}
Score: {score}/100
Status: {status}
Action: {action}
Reasons: {', '.join(reasons) if reasons else 'Standard scoring'}"""

        return [types.TextContent(type="text", text=result)]

    # ── Study Progress ─────────────────────────────
    elif name == "get_study_progress":
        day = arguments["current_day"]
        total_days = 168
        hours_per_day = 6
        hours_done = day * hours_per_day
        total_hours = total_days * hours_per_day
        percent = round((day / total_days) * 100, 1)
        days_left = total_days - day

        phases = []
        if day >= 1:
            phases.append("✅ Phase 1: AI Foundations (Days 1-14)")
        if day >= 15:
            phases.append("✅ Phase 2: n8n Automation (Days 15-42)")
        if day >= 43:
            phases.append("✅ Phase 3: Python (Days 43-84)")
        if day >= 85:
            phases.append("🔄 Phase 4: AI Agents (Days 85-126)")
        if day >= 127:
            phases.append("🔄 Phase 5: Expert Level (Days 127-168)")

        result = f"""Abdullah's Study Progress:
📅 Day: {day} of {total_days}
📊 Progress: {percent}%
⏱️  Hours studied: {hours_done} of {total_hours}
📆 Days remaining: {days_left}
⏰ Hours remaining: {days_left * hours_per_day}

Phases:
{chr(10).join(phases)}

Keep going! You are {percent}% of the way to expert level."""

        return [types.TextContent(type="text", text=result)]

    # ── Report Formatter ───────────────────────────
    elif name == "format_automation_report":
        title = arguments["title"]
        data = arguments["data"]

        timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M")

            f"{'='*50}",

    else:
        return [types.TextContent(
            type="text",
            text=f"Unknown tool: {name}"

from langchain_groq import ChatGroq
from langchain_core.messages import HumanMessage, ToolMessage

llm = ChatGroq(model="openai/gpt-oss-20b", temperature=0, api_key=GROQ_API_KEY)
llm_with_tools = llm.bind_tools(tools_schema)

def run_agent(question: str) -> str:
    messages = [HumanMessage(content=question)]
    for _ in range(6):  # max iterations
        ai_msg = llm_with_tools.invoke(messages)
        messages.append(ai_msg)
        if not ai_msg.tool_calls:
            return ai_msg.content
        for call in ai_msg.tool_calls:
            result = run_mcp_tool(call["name"], call["args"])
            messages.append(ToolMessage(content=result, tool_call_id=call["id"]))
    return "Max iterations reached."

    @app.route('/mcp/call', methods=['POST'])
    def call_mcp():
        data = request.json
        if not data:
            return jsonify({"error": "No data provided"}), 400
        tool_name = data.get("tool")
        arguments = data.get("arguments", {})
        if not tool_name:
            return jsonify({"error": "No tool specified"}), 400
        try:
            result = run_mcp_tool(tool_name, arguments)
            return jsonify({
                "success": True, "tool": tool_name, "result": result,
                "timestamp": datetime.datetime.now().isoformat()
            })
        except Exception as e:
            return jsonify({"success": False, "error": str(e)}), 500

    @app.route('/mcp/resources', methods=['GET'])
    def list_resources():
        return jsonify({"resources": [
            {"uri": "automation://learning-progress", "name": "Learning Progress", "type": "json"},
            {"uri": "automation://tool-stack", "name": "Free Tool Stack", "type": "json"}
        ]})

    @app.route('/mcp/resource', methods=['POST'])
    def get_resource():
        data = request.json
        uri = data.get("uri", "")
        if uri == "automation://learning-progress":
            return jsonify({"uri": uri, "data": {
                "learner": "Abdullah", "current_day": 23, "total_days": 168,
                "current_phase": "Phase 4 — AI Agents",
                "progress_percent": round(23/168*100, 1)
            }})
        elif uri == "automation://tool-stack":
            return jsonify({"uri": uri, "data": {
                "automation": "n8n", "ai_api": "Groq", "language": "Python",
                "framework": "LangChain + CrewAI", "vector_store": "FAISS"
            }})
        return jsonify({"error": "Resource not found"}), 404

    @app.route('/health', methods=['GET'])
    def health():
        return jsonify({"status": "MCP server running", "version": "1.0.0"})

    def run_mcp_http():
        app.run(port=5001, debug=False, use_reloader=False, threaded=True)

    thread = threading.Thread(target=run_mcp_http)
    thread.daemon = True
    thread.start()
